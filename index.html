📋 FIRESTORE-REGLER (kopiera & klistra in)

Gå till Firebase Console → Firestore → Rules och ersätt med dessa regler:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Bokningar — publikt läsbar, skyddad skrivning
    match /bookings/{seat} {
      allow read: if true;
      allow create: if request.resource.data.name is string
        && request.resource.data.discord is string
        && request.resource.data.seat == seat
        && request.resource.data.name.size() <= 60
        && request.resource.data.discord.size() <= 60;
      allow update, delete: if request.auth != null;
    }

    // Anmälningar — publikt skapa, admin läser/ändrar/raderar
    match /registrations/{doc} {
      allow read, delete, update: if request.auth != null;
      allow create: if request.resource.data.name is string
        && request.resource.data.email is string
        && request.resource.data.name.size() <= 100;
    }

    // Kontaktmeddelanden — bara skapa
    match /messages/{doc} {
      allow create: if request.resource.data.name is string
        && request.resource.data.message is string
        && request.resource.data.message.size() <= 1000;
      allow read, update, delete: if request.auth != null;
    }

    // Config — publikt läsbar, bara inloggad admin skriver
    match /config/{doc} {
      allow read: if true;
      allow write: if request.auth != null;
    }

  }
}
